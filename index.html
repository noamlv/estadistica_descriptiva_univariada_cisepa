4.1 4.1 Funcion para moda y variacion modal

def estadisticas_nominales(dataframe, columna):
    datos = dataframe[columna].dropna()
    total = len(datos)
    freq = Counter(datos)
    moda = freq.most_common(1)[0]
    frecuencia_moda = moda[1]
    variacion_modal = 1 - (frecuencia_moda / total) if total > 0 else 0

    tabla = pd.DataFrame({
        "categoria": list(freq.keys()),
        "frecuencia": list(freq.values()),
        "frec_relativa": [f / total for f in freq.values()],
        "frec_acumulada": np.cumsum(list(freq.values())) / total
    }).sort_values("frecuencia", ascending=False)

    return {
        "variable": columna,
        "moda": moda[0],
        "frecuencia_moda": moda[1],
        "variacion_modal": round(variacion_modal, 4),
        "n_categorias": len(freq),
        "tabla": tabla
    }

5.1 5.1 Funcion completa de medidas

def estadisticas_cuantitativas(dataframe, columna):
    datos = dataframe[columna].dropna()
    if len(datos) == 0:
        return None

    # Centralizacion
    media = datos.mean()
    mediana = datos.median()
    moda = stats.mode(datos, keepdims=True).mode[0]

    # Posicion
    q1 = np.percentile(datos, 25)
    q3 = np.percentile(datos, 75)
    percentiles = {
        "P10": np.percentile(datos, 10),
        "P25": q1, "P50": mediana,
        "P75": q3, "P90": np.percentile(datos, 90)
    }

    # Dispersion
    rango = datos.max() - datos.min()
    varianza = datos.var()
    desviacion = datos.std()
    cv = (desviacion / media * 100) if media != 0 else np.nan
    iqr = q3 - q1

    # Simetria
    asimetria_pearson = stats.skew(datos)
    asimetria_yules = (q3 + q1 - 2 * mediana) / (q3 - q1) if (q3 - q1) != 0 else np.nan

    # Apuntamiento
    curtosis_fisher = stats.kurtosis(datos)

    return {
        "variable": columna,
        "n": len(datos),
        "nulos": dataframe[columna].isna().sum(),
        "media": round(media, 4),
        "mediana": round(mediana, 4),
        "moda": moda,
        "percentiles": percentiles,
        "rango": round(rango, 4),
        "varianza": round(varianza, 4),
        "de": round(desviacion, 4),
        "cv": round(cv, 4),
        "iqr": round(iqr, 4),
        "asimetria_pearson": round(asimetria_pearson, 4),
        "asimetria_yules": round(asimetria_yules, 4),
        "curtosis_fisher": round(curtosis_fisher, 4)
    }

8 8. Mapa de concentracion: Gini

def gini(x):
    x = np.array(x, dtype=float)
    x = x[~np.isnan(x)]
    if len(x) == 0 or x.sum() == 0:
        return np.nan
    x_sorted = np.sort(x)
    n = len(x)
    index = np.arange(1, n + 1)
    return (2 * np.sum(index * x_sorted) / (n * np.sum(x_sorted))) - (n + 1) / n
